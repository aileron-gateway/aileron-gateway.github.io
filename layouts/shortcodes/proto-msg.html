{{- $url := .Get "url" -}} {{- $msg := .Get "msg" -}} {{ $lang := .Get "lang" }} {{ $opts := .Get "opts" }} {{- $remote := resources.GetRemote $url -}} {{- $content := $remote.Content -}} {{- $lines := split $content "\n" -}} {{- $startTag := trim (printf "message %s {" $msg) " " -}} {{- $endTag := "}" -}} {{- $.Scratch.Set "startl" -1 -}} {{- $.Scratch.Set "endl" -1 -}} {{- range $index, $line := $lines -}} {{- if hasPrefix $line $startTag -}} {{- $.Scratch.Set "startl" $index -}} {{- else if hasPrefix $line $endTag -}} {{- $startl := $.Scratch.Get "startl" -}} {{- $endl := $.Scratch.Get "endl" -}} {{- if and (gt $startl 0) (lt $endl 0) -}} {{- $.Scratch.Set "endl" $index -}} {{- end -}} {{- end -}} {{- end -}} {{- $startl := $.Scratch.Get "startl" -}} {{- $endl := $.Scratch.Get "endl" -}} {{- if lt $startl 0 -}} {{- errorf "Named snippet is missing START tag: %s" $startTag -}} {{- end -}} {{- if lt $endl 0 -}} {{- errorf "Named snippet is missing END tag: %s" $endTag -}} {{- end -}} {{- $snippetLen := sub $endl $startl -}} {{- $includedLines := first $snippetLen (after (add $startl 1) $lines) -}} {{- $snippet := delimit $includedLines "\n" -}} {{- highlight (trim $snippet "\n\r") $lang $opts -}}
